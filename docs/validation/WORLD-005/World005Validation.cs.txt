using System;
using System.IO;
using System.Reflection;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.InputSystem;
using UnityEngine.InputSystem.LowLevel;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;
using RTSWorld.Presentation.Cameras;
using RTSWorld.Presentation.Streaming;

[InitializeOnLoad]
public static class World005Validation
{
    const string Key="WORLD005Validation";
    const string Output="/tmp/rtswater-unity-validation/world005-captures";
    static readonly string[] Names={"initial","wasd","middle-drag","zoom-min","zoom-max","rotate","edge-pan","edge-disabled","water-seed7","hills-min-zoom","auto-update-idle"};
    static int ticks; static bool idlePrepared; static RenderTexture target;
    static Keyboard keyboard; static Mouse mouse;
    static World005Validation()
    {
        EditorApplication.update+=Update;
        Application.logMessageReceived+=(message,trace,type)=>{
            if(SessionState.GetBool(Key,false)&&(type==LogType.Error||type==LogType.Exception||type==LogType.Assert))
                File.AppendAllText(Output+"/errors.txt",message+"\n"+trace+"\n");
        };
    }
    public static void Run()
    {
        Directory.CreateDirectory(Output);
        EditorSceneManager.OpenScene("Assets/_Project/Scenes/RTSCameraDemo.unity");
        SessionState.SetBool(Key,true);SessionState.SetInt(Key+"stage",0);SessionState.SetInt(Key+"index",0);ticks=0;
    }
    static void Update()
    {
        if(!SessionState.GetBool(Key,false)||EditorApplication.isCompiling||EditorApplication.isUpdating||++ticks<100)return;
        ticks=0;
        try
        {
            int stage=SessionState.GetInt(Key+"stage",0),index=SessionState.GetInt(Key+"index",0);
            if(stage==2){if(EditorApplication.isPlaying)return;SessionState.SetBool(Key,false);EditorApplication.Exit(0);return;}
            if(stage==1&&!EditorApplication.isPlaying)return;
            if(index==Names.Length){
                if(stage==0){SessionState.SetInt(Key+"stage",1);SessionState.SetInt(Key+"index",0);EditorApplication.EnterPlaymode();}
                else{SessionState.SetInt(Key+"stage",2);EditorApplication.ExitPlaymode();}
                return;
            }
            var controller=UnityEngine.Object.FindAnyObjectByType<RtsCameraController>();
            var bridge=UnityEngine.Object.FindAnyObjectByType<CameraStreamingObserver>();
            var reader=UnityEngine.Object.FindAnyObjectByType<RtsCameraInput>();
            var manager=UnityEngine.Object.FindAnyObjectByType<ChunkManager>();
            var surface=UnityEngine.Object.FindAnyObjectByType<WorldSurfaceQuery>();
            var camera=controller.OutputCamera;
            if(!idlePrepared)
            {
                if(keyboard==null||!keyboard.added)keyboard=InputSystem.AddDevice<Keyboard>();
                if(mouse==null||!mouse.added)mouse=InputSystem.AddDevice<Mouse>();
                controller.enabled=false;bridge.enabled=false;reader.InputEnabled=true;
                target=new RenderTexture(1600,1000,24,RenderTextureFormat.ARGB32);target.Create();camera.targetTexture=target;
                manager.Seed=index==8?7:12345;
                var settings=new RtsCameraSettings();if(index==7)settings.EnableEdgePan=false;
                controller.Configure(camera,reader,surface,settings,index==8?64:index==9?104:0,index==8?60:index==9?-32:0,index==8?18:index==9?12:24);
                void Step(KeyboardState keys,MouseState pointer){
                    InputSystem.QueueStateEvent(keyboard,keys);InputSystem.QueueStateEvent(mouse,pointer);InputSystem.Update();
                    controller.Advance(reader.Read(settings,true,new Vector2(1600,1000)),1.0/60,1000);bridge.Sync();
                }
                var center=new MouseState{position=new Vector2(800,500)};
                Step(new KeyboardState(),center);
                if(index==1)for(int i=0;i<120;i++)Step(new KeyboardState(UnityEngine.InputSystem.Key.W,UnityEngine.InputSystem.Key.D),center);
                if(index==2)for(int i=0;i<=60;i++)Step(new KeyboardState(),new MouseState{position=new Vector2(800+i*4,500+i*2),delta=new Vector2(4,2)}.WithButton(MouseButton.Middle));
                if(index==3||index==4){for(int i=0;i<10;i++)Step(new KeyboardState(),new MouseState{position=new Vector2(800,500),scroll=new Vector2(0,index==3?120:-120)});for(int i=0;i<180;i++)Step(new KeyboardState(),center);}
                if(index==5)for(int i=0;i<60;i++)Step(new KeyboardState(UnityEngine.InputSystem.Key.E),center);
                if(index==6||index==7)for(int i=0;i<120;i++)Step(new KeyboardState(),new MouseState{position=new Vector2(1599,500)});
                if(index==7&&(Math.Abs(controller.Motion.FocusX)>1e-6||Math.Abs(controller.Motion.FocusZ)>1e-6))throw new Exception("Disabled edge pan moved the focus");
                bridge.Sync();
                for(int i=0;i<400;i++){manager.Step();if(manager.PendingCount==0)break;}
                if(manager.PendingCount!=0||bridge.CoverageLimited)throw new Exception("Streaming coverage did not stabilize");
                if(index==10){reader.InputEnabled=false;controller.enabled=true;bridge.enabled=true;idlePrepared=true;return;}
            }
            var points=new Vector3[4];camera.CalculateFrustumCorners(new Rect(0,0,1,1),camera.nearClipPlane,UnityEngine.Camera.MonoOrStereoscopicEye.Mono,points);
            float lowest=float.MaxValue;foreach(var p in points)lowest=Math.Min(lowest,camera.transform.TransformPoint(p).y);
            if(lowest<surface.MaximumHeight+controller.Settings.TerrainClearance-0.001f)throw new Exception("Near plane intersects the terrain bound");
            RenderPipeline.SubmitRenderRequest(camera,new UniversalRenderPipeline.SingleCameraRequest{destination=target});
            var previous=RenderTexture.active;RenderTexture.active=target;
            var texture=new Texture2D(target.width,target.height,TextureFormat.RGB24,false);texture.ReadPixels(new Rect(0,0,target.width,target.height),0,0);texture.Apply();
            File.WriteAllBytes(Output+"/"+(stage==0?"editor-":"play-")+Names[index]+".png",texture.EncodeToPNG());
            RenderTexture.active=previous;UnityEngine.Object.DestroyImmediate(texture);camera.targetTexture=null;target.Release();UnityEngine.Object.DestroyImmediate(target);target=null;
            var flags=BindingFlags.Static|BindingFlags.Public|BindingFlags.NonPublic;var logType=typeof(Editor).Assembly.GetType("UnityEditor.LogEntries");var counts=new object[]{0,0,0};logType.GetMethod("GetCountsByType",flags).Invoke(null,counts);
            File.AppendAllText(Output+"/metrics.txt",$"{(stage==0?"Editor":"Play")} {Names[index]} focus=({controller.Motion.FocusX:F6},{controller.Motion.FocusZ:F6}) zoom={controller.Motion.Zoom:F4} yaw={controller.Motion.Yaw:F2} pitch={controller.Settings.Inclination:F2} radius={manager.LoadRadius} chunks={manager.ActiveCount} pending={manager.PendingCount} minNearY={lowest:F4} heightBound={surface.MaximumHeight:F4} shaderError={ShaderUtil.ShaderHasError(manager.EnvironmentLayer.Material.shader)} errors={counts[0]} warnings={counts[1]} autoUpdate={controller.enabled}\n");
            if(index==10&&stage==1){
                for(int i=0;i<32;i++){controller.Advance(default,1.0/60,1000);bridge.Sync();}
                var watch=new System.Diagnostics.Stopwatch();long bytes=GC.GetAllocatedBytesForCurrentThread();watch.Start();
                for(int i=0;i<1000;i++){controller.Advance(default,1.0/60,1000);bridge.Sync();}
                watch.Stop();long allocated=GC.GetAllocatedBytesForCurrentThread()-bytes;
                File.WriteAllText(Output+"/hotpath.txt",$"1000 warmed stationary controller+surface+observer calls, totalMs={watch.Elapsed.TotalMilliseconds:F4}, allocatedBytes={allocated}; Stopwatch created before allocation baseline; no rendering, GUI or ChunkManager.Step in this measurement.\n");
            }
            idlePrepared=false;SessionState.SetInt(Key+"index",index+1);
        }
        catch(Exception e){File.AppendAllText(Output+"/errors.txt",e+"\n");SessionState.SetBool(Key,false);EditorApplication.Exit(1);}
    }
}
